export function streetGroundHeight(x,z){
 if((x>=8&&x<=16&&z>=11&&z<=17)||(x>=-25&&x<=-10&&z>=-5.5&&z<=1))return .17;
 if(x>=21&&x<=35&&z>=21.5&&z<=34.5)return .18;
 if(Math.abs(Math.abs(x)-5.15)<.13&&(z< -14.7||z> -5.3))return .245;
 if((Math.abs(x)>=5.15&&Math.abs(x)<=8.85&&Math.abs(z)<48)||(Math.abs(x)<45.5&&(Math.abs(z+15.7)<1.5||Math.abs(z+4.3)<1.5)))return .17;
 if(Math.abs(x)<5||Math.abs(z+10)<4)return .085;
 return -.105;
}
